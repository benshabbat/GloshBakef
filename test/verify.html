בדיקת כתובת אמיתית

העתק כתובת של תמונה מאתר אמיתי (באינסטגרם: לחיצה ימנית על תמונה בפיד ← «העתק כתובת תמונה») והדבק כאן. הדף ידווח באיזה מסלול התוסף הלך ואם הקנבס נגוע — כלומר אם מסלול הפיקסלים חסום באתר הזה.

—

בדיקה בתוך האתר עצמו

אתרים גדולים חותמים את כתובות התמונות שלהם וקושרים אותן להקשר, ולכן כתובת שמודבקת בדף חיצוני עלולה פשוט לא להיטען — וזה לא אומר כלום על התוסף. הבדיקה האמינה רצה על התמונה החיה במקום שבו היא נמצאת.

פתח את האתר, גלול עד שנטענו תמונות, פתח DevTools על הדף עצמו (F12 ← Console) והדבק: 

(async () => {
  const big = [...document.images].filter(i => i.naturalWidth >= 200);
  if (!big.length) { console.log("לא נמצאו תמונות גדולות — גלול ונסה שוב"); return; }
  const img = big[0];
  const out = {
    "תמונות גדולות בדף": big.length,
    "crossorigin": img.crossOrigin ?? "(לא מוגדר)",
    "פסק דין": img.getAttribute("data-imgfilter") ?? "(לא תויג כלל)",
    "גודל טבעי": img.naturalWidth + "x" + img.naturalHeight
  };
  try {
    const c = Object.assign(document.createElement("canvas"), { width: 224, height: 224 });
    c.getContext("2d").drawImage(img, 0, 0, 224, 224);
    c.toDataURL("image/jpeg", 0.75);
    out["קנבס"] = "נקי — מסלול הפיקסלים עובד";
  } catch (e) {
    out["קנבס"] = "נגוע (" + e.name + ") — מסלול הפיקסלים חסום";
  }
  out["מצבים בדף"] = [...document.querySelectorAll("[data-imgfilter]")]
    .reduce((a, el) => { const s = el.getAttribute("data-imgfilter"); a[s] = (a[s] ?? 0) + 1; return a; }, {});
  console.log(out);
  console.log("הכתובת לבדיקת ה-service worker:", img.currentSrc);
})();
